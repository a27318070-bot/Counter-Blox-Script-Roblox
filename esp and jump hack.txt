-- ВНИМАНИЕ: Это образовательный пример. Использование читов нарушает правила Roblox

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

-- Состояние ESP
local ESP_ENABLED = false
local ESPHolders = {}

-- Функция создания ESP для игрока
local function createESP(player)
	if not ESP_ENABLED then return end

	-- Удаляем старый ESP если есть
	if ESPHolders[player] then
		if ESPHolders[player].Highlight then
			ESPHolders[player].Highlight:Destroy()
		end
		if ESPHolders[player].Billboard then
			ESPHolders[player].Billboard:Destroy()
		end
	end

	local character = player.Character
	if not character then return end

	-- Ждем полной загрузки персонажа
	if not character:FindFirstChild("Humanoid") then
		character:WaitForChild("Humanoid", 5)
	end

	if not character:FindFirstChild("HumanoidRootPart") then
		character:WaitForChild("HumanoidRootPart", 5)
	end

	if not character.Humanoid or not character.HumanoidRootPart then return end

	-- Создаем Highlight для подсветки
	local highlight = Instance.new("Highlight")
	highlight.Name = "ESP_Highlight"
	highlight.Adornee = character
	highlight.FillColor = Color3.fromRGB(255, 0, 0)
	highlight.OutlineColor = Color3.fromRGB(255, 255, 0)
	highlight.FillTransparency = 0.7
	highlight.OutlineTransparency = 0
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.Parent = character

	-- Создаем BillboardGui для отображения имени
	local billboard = Instance.new("BillboardGui")
	billboard.Name = "ESP_Billboard"
	billboard.Size = UDim2.new(0, 200, 0, 50)
	billboard.AlwaysOnTop = true
	billboard.StudsOffset = Vector3.new(0, 3, 0)
	billboard.Adornee = character.HumanoidRootPart
	billboard.MaxDistance = 500
	billboard.Parent = character.HumanoidRootPart

	local label = Instance.new("TextLabel")
	label.Text = player.Name
	label.TextColor3 = Color3.fromRGB(255, 255, 255)
	label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	label.TextStrokeTransparency = 0
	label.TextSize = 14
	label.Font = Enum.Font.GothamBold
	label.BackgroundTransparency = 1
	label.Size = UDim2.new(1, 0, 1, 0)
	label.Parent = billboard

	-- Сохраняем ссылки
	ESPHolders[player] = {
		Highlight = highlight,
		Billboard = billboard,
		Character = character
	}

	print("ESP создан для: " .. player.Name)
end

-- Функция удаления ESP
local function removeESP(player)
	if ESPHolders[player] then
		if ESPHolders[player].Highlight then
			ESPHolders[player].Highlight:Destroy()
		end
		if ESPHolders[player].Billboard then
			ESPHolders[player].Billboard:Destroy()
		end
		ESPHolders[player] = nil
		print("ESP удален для: " .. player.Name)
	end
end

-- Функция включения/выключения ESP
local function toggleESP()
	ESP_ENABLED = not ESP_ENABLED

	if ESP_ENABLED then
		-- Включаем ESP для всех игроков
		for _, player in pairs(Players:GetPlayers()) do
			if player ~= LocalPlayer then
				spawn(function()
					createESP(player)
				end)
			end
		end
		print("ESP ВКЛЮЧЕН")
	else
		-- Выключаем ESP для всех игроков
		for player, esp in pairs(ESPHolders) do
			removeESP(player)
		end
		ESPHolders = {}
		print("ESP ВЫКЛЮЧЕН")
	end
end

-- Обработчик для новых игроков
local function onPlayerAdded(player)
	if player == LocalPlayer then return end

	print("Игрок добавлен: " .. player.Name)

	-- Обработка появления персонажа
	local function onCharacterAdded(character)
		print("Персонаж добавлен для: " .. player.Name)

		-- Ждем полной загрузки персонажа
		local humanoid = character:WaitForChild("Humanoid", 10)
		local rootPart = character:WaitForChild("HumanoidRootPart", 10)

		if humanoid and rootPart then
			-- Создаем ESP после загрузки
			wait(1) -- Даем время на полную загрузку
			if ESP_ENABLED then
				createESP(player)
			end

			-- Отслеживаем смерть
			humanoid.Died:Connect(function()
				print("Персонаж умер: " .. player.Name)
				removeESP(player)

				-- Ждем респавн
				humanoid:GetPropertyChangedSignal("Health"):Wait()
				if humanoid.Health > 0 then
					wait(2) -- Даем время на респавн
					if ESP_ENABLED then
						createESP(player)
					end
				end
			end)
		end
	end

	-- Подключаем обработчик персонажа
	if player.Character then
		onCharacterAdded(player.Character)
	end
	player.CharacterAdded:Connect(onCharacterAdded)

	-- Обработка выхода игрока
	player.AncestryChanged:Connect(function()
		if not player:IsDescendantOf(Players) then
			removeESP(player)
		end
	end)
end

-- Обработчик для вышедших игроков
local function onPlayerRemoving(player)
	print("Игрок вышел: " .. player.Name)
	removeESP(player)
end

-- Инициализация для существующих игроков
for _, player in pairs(Players:GetPlayers()) do
	if player ~= LocalPlayer then
		onPlayerAdded(player)
	end
end

-- Подписка на события
Players.PlayerAdded:Connect(onPlayerAdded)
Players.PlayerRemoving:Connect(onPlayerRemoving)


-- Обработчик нажатия кнопки
esp.MouseButton1Click:Connect(function()
	toggleESP()

	-- Обновляем текст кнопки
	if ESP_ENABLED then
		esp.Text = "ESP: ON"
		esp.BackgroundColor3 = Color3.fromRGB(0, 100, 0)
	else
		esp.Text = "ESP: OFF"
		esp.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
	end
end)

-- Функция для принудительного обновления всех ESP
local function refreshAllESP()
	print("Принудительное обновление ESP...")
	if ESP_ENABLED then
		for _, player in pairs(Players:GetPlayers()) do
			if player ~= LocalPlayer then
				spawn(function()
					createESP(player)
				end)
			end
		end
	end
end

-- Автоочистка при выходе
LocalPlayer.AncestryChanged:Connect(function()
	if not LocalPlayer:IsDescendantOf(Players) then
		for player, esp in pairs(ESPHolders) do
			removeESP(player)
		end
		ESPHolders = {}
	end
end)

-- Добавляем команду в консоль для отладки
_G.ESP = {
	toggle = toggleESP,
	refresh = refreshAllESP,
	status = function() return ESP_ENABLED end
}

print("ESP система загружена!")
print("Используйте: _G.ESP.toggle() - включить/выключить")
print("Используйте: _G.ESP.refresh() - обновить ESP")
print("Используйте: _G.ESP.status() - проверить статус")
